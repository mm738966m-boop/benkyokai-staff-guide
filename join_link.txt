/**
 * お金の勉強会「入室リンク」
 * UTAGEの予約通知メール（[FXBOOK]…[/FXBOOK]）を台帳シートに記録し、
 * Webアプリ（?m=メールアドレス）で、その人の次の予約のZoom URLを返す。
 * 見つからないときは DEFAULT_ZOOM（これまでの固定の部屋）を返す。
 * 初回：setup() を1回実行 → デプロイ（ウェブアプリ・全員）
 */
const DEFAULT_ZOOM = 'https://us02web.zoom.us/j/86141509775';
const SHEET_NAME = '台帳';
const HEAD = ['受信日時', '申込番号', 'メール', 'お名前', '開始', '終了', '担当', 'Zoom', '状態', 'メッセージID'];

function setup() {
  const props = PropertiesService.getScriptProperties();
  if (!props.getProperty('SHEET_ID')) {
    const ss = SpreadsheetApp.create('勉強会_入室リンク台帳');
    const sh = ss.getSheets()[0];
    sh.setName(SHEET_NAME);
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    props.setProperty('SHEET_ID', ss.getId());
  }
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'ingest') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('ingest').timeBased().everyMinutes(5).create();
  ingest();
  Logger.log(SpreadsheetApp.openById(props.getProperty('SHEET_ID')).getUrl());
}

function sheet_() {
  return SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('SHEET_ID')).getSheetByName(SHEET_NAME);
}

function block_(body, tag) {
  const m = body.match(new RegExp('\\[' + tag + '\\]([\\s\\S]*?)\\[/' + tag + '\\]'));
  if (!m) return null;
  const o = {};
  m[1].split(/\r?\n/).forEach(function (line) {
    const i = line.indexOf('=');
    if (i > 0) o[line.slice(0, i).trim()] = line.slice(i + 1).trim();
  });
  return o;
}

function toDate_(dateStr, timeStr) {
  const d = String(dateStr).match(/(\d{4})\D+(\d{1,2})\D+(\d{1,2})/);
  const t = String(timeStr).match(/(\d{1,2}):(\d{2})/);
  if (!d || !t) return null;
  return new Date(+d[1], +d[2] - 1, +d[3], +t[1], +t[2]);
}

function ingest() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return;
  try {
    const sh = sheet_();
    const last = sh.getLastRow();
    const rows = last > 1 ? sh.getRange(2, 1, last - 1, HEAD.length).getValues() : [];
    const seen = {};
    rows.forEach(function (r) { seen[r[9]] = true; });
    const threads = GmailApp.search('from:no-reply@utage-notify.com subject:(勉強会予約 OR 勉強会キャンセル) newer_than:14d', 0, 50);
    const msgs = [];
    threads.forEach(function (th) { th.getMessages().forEach(function (m) { msgs.push(m); }); });
    msgs.sort(function (a, b) { return a.getDate() - b.getDate(); });
    msgs.forEach(function (m) {
      const id = m.getId();
      if (seen[id]) return;
      const body = m.getPlainBody();
      const b = block_(body, 'FXBOOK');
      const c = block_(body, 'FXCANCEL');
      if (b && b.mail) {
        const s = toDate_(b.date, b.start), e = toDate_(b.date, b.end);
        sh.appendRow([m.getDate(), b.id || '', b.mail.toLowerCase(), b.name || '', s || b.date + ' ' + b.start, e || '', b.member || '', b.zoom || '', '予約', id]);
        seen[id] = true;
      } else if (c && (c.id || c.mail)) {
        const n = sh.getLastRow();
        const cur = n > 1 ? sh.getRange(2, 1, n - 1, HEAD.length).getValues() : [];
        const cs = toDate_(c.date, c.start);
        for (let i = cur.length - 1; i >= 0; i--) {
          const same = (c.id && String(cur[i][1]) === String(c.id)) ||
            (!c.id && cur[i][2] === String(c.mail).toLowerCase() && cs && cur[i][4] instanceof Date && cur[i][4].getTime() === cs.getTime());
          if (same && cur[i][8] === '予約') { sh.getRange(i + 2, 9).setValue('キャンセル'); break; }
        }
        sh.appendRow([m.getDate(), c.id || '', String(c.mail || '').toLowerCase(), '', cs || '', '', '', '', 'キャンセル通知', id]);
        seen[id] = true;
      }
    });
  } finally {
    lock.releaseLock();
  }
}

function find_(mail) {
  const sh = sheet_();
  const last = sh.getLastRow();
  if (last < 2) return null;
  const rows = sh.getRange(2, 1, last - 1, HEAD.length).getValues();
  const now = Date.now();
  const cands = [];
  rows.forEach(function (r) {
    if (r[2] !== mail || r[8] !== '予約' || !r[7]) return;
    const s = r[4] instanceof Date ? r[4].getTime() : null;
    const e = r[5] instanceof Date ? r[5].getTime() : (s ? s + 3600000 : null);
    if (!s || e + 30 * 60000 < now) return;
    cands.push({ s: s, got: r[0] instanceof Date ? r[0].getTime() : 0, url: r[7] });
  });
  if (!cands.length) return null;
  // もうすぐ始まる（90分以内）予約があればそれ。なければ、いちばん新しく入った予約（取り直しに対応）
  const soon = cands.filter(function (c) { return c.s - now <= 90 * 60000; }).sort(function (x, y) { return x.s - y.s; });
  if (soon.length) return soon[0];
  cands.sort(function (x, y) { return y.got - x.got; });
  return cands[0];
}

function doGet(e) {
  const p = (e && e.parameter) || {};
  const mail = String(p.m || '').trim().toLowerCase();
  let hit = null;
  if (mail && mail.indexOf('@') > 0 && mail.indexOf('%') === -1) {
    hit = find_(mail);
    if (!hit) { try { ingest(); } catch (err) {} hit = find_(mail); }
  }
  const out = JSON.stringify({ url: hit ? hit.url : DEFAULT_ZOOM, found: !!hit });
  if (p.callback && /^[A-Za-z0-9_]+$/.test(p.callback)) {
    return ContentService.createTextOutput(p.callback + '(' + out + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}
